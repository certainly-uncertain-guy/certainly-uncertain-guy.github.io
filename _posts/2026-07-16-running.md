  // If the Chart.js CDN is unreachable, skip the charts rather than throwing a
  // ReferenceError. The hero stats and the SVG map do not depend on it and
  // still render.
  if (typeof Chart === 'undefined') {
    console.error('Chart.js failed to load; skipping running charts.');
    return;
  }
  RunningCharts.loadData(function (records) {
    RunningCharts.renderBuildupChart(records);
    RunningCharts.renderMonthlyMileageChart(records);
    RunningCharts.renderPaceTrendChart(records);
    RunningCharts.renderHrTrendChart(records);
    RunningCharts.renderCadenceTrendChart(records);
  });
});
</script>
